```go
package job

import "strings"

// 系統 index 黑名單
var systemIndexPrefixes = []string{
	".",                  // 所有以 . 開頭的 index (.kibana, .security, etc.)
	"ilm-history-",       // ILM 歷史記錄
	"kibana_sample_",     // Kibana 範例數據
}

// isSystemIndex 檢查是否為系統 index
func isSystemIndex(indexName string) bool {
	for _, prefix := range systemIndexPrefixes {
		if strings.HasPrefix(indexName, prefix) {
			return true
		}
	}
	return false
}
```

```go
func FilterType_pattern(kind string, value []string) (indiceslist []string) {
	indicesinfo := CatIndices()
	var indices []string

	if kind == "prefix" {
		for data := range indicesinfo {
			// ✅ 先檢查是否為系統 index
			if isSystemIndex(indicesinfo[data].Index) {

				if err != nil {
					global.Logger.Error(err.Error())
				}
				if matchbool {
					indices = append(indices, indicesinfo[data].Index)
				}
			}
		}
	} else if kind == "suffix" {
		// ... 同樣添加 isSystemIndex 檢查
	} else if kind == "regex" {
		// ... 同樣添加 isSystemIndex 檢查
	}

	// ✅ 新增：檢查是否有匹配
	if len(indices) == 0 {
		global.Logger.Warnw("⚠️ Pattern filter matched NO indices (system indices excluded)",
			"kind", kind,
			"patterns", value)
	} else {
		global.Logger.Infow("Pattern filter matched indices",
			"kind", kind,
			"patterns", value,
			"matched_count", len(indices))
```

#### 修復 1.6：添加 delete_indices 保護

```go
func handleDeleteIndices(uuid string, indexList []string) {
	// ✅ 安全檢查：系統 index 保護
	systemIndices := []string{}
	for _, idx := range indexList {
		if isSystemIndex(idx) {
			systemIndices = append(systemIndices, idx)
		}
	}

	if len(systemIndices) > 0 {
		global.Logger.Errorw("🚨 CRITICAL: Attempting to delete SYSTEM indices - OPERATION ABORTED",
			"system_indices", systemIndices,
			"uuid", uuid,
			"action", "delete_indices_blocked")
		return
	}

	// ✅ 安全檢查：索引數量警告
	if len(indexList) > 100 {
		global.Logger.Warnw("⚠️ Large deletion operation detected",
			"index_count", len(indexList),
			"uuid", uuid,
			"warning", "Deleting more than 100 indices")
	}

	executor := &ActionExecutor{UUID: uuid, Action: "delete_indices"}
	if indexList != nil {
		executor.LogActionSummary("delete_indices", indexList)
		operation := func(indices []string) {
			DeleteIndex(indices)
		}
		executor.ExecuteOnIndices(indexList, operation)
	}
}
```

```go
// validateFilters 驗證過濾器配置
func validateFilters(actionIndex int, action structs.Actiond) error {
	hasAge := false
	hasSpace := false
	hasWaterLevel := false
	hasPattern := false

	// ... (原有的 filter 驗證邏輯) ...

	// 檢查互斥過濾器（原有邏輯）
	if hasAge && hasSpace {
		return fmt.Errorf("action[%d]: age 和 space 過濾器不能同時使用", actionIndex)
	}
	if hasAge && hasWaterLevel {
		return fmt.Errorf("action[%d]: age 和 water_level 過濾器不能同時使用", actionIndex)
	}
	if hasSpace && hasWaterLevel {
		return fmt.Errorf("action[%d]: space 和 water_level 過濾器不能同時使用", actionIndex)
	}

	// 🔒 P1-2.1 修復：強制性安全檢查（新增）

	// 1. delete_indices 必須有 pattern filter（防止意外刪除系統 index）
	if action.Action == "delete_indices" && !hasPattern {
		return fmt.Errorf("action[%d]: delete_indices action 必須配置 pattern filter 以避免意外刪除系統索引", actionIndex)
	}

	// 2. space/water_level 必須配合 pattern filter（防止選中所有 index）
	if (hasSpace || hasWaterLevel) && !hasPattern {
		filterType := "space"
		if hasWaterLevel {
			filterType = "water_level"
		}
		return fmt.Errorf("action[%d]: %s filter 必須配合 pattern filter 使用，否則會影響所有索引（包括系統索引）", actionIndex, filterType)
	}

	// 3. 非 rollover action 建議有 pattern 過濾器（安全措施 - 僅警告）
	if action.Action != "rollover" && !hasPattern && action.Action != "delete_indices" {
		if global.Logger != nil {
			global.Logger.Warnw("⚠️  建議為所有非 rollover action 添加 pattern 過濾器以避免意外操作系統索引",
				"action_index", actionIndex,
				"action_type", action.Action)
		}
	}

	return nil
}
```

## 📖 給使用者的安全建議

### ✅ 安全的配置範例

```yaml
# 範例 1：安全的刪除操作
- action: delete_indices
  description: "Delete old test logs"
  filters:
    - filtertype: pattern
      kind: prefix
      value: logs-test-      # ✅ 明確指定前綴
    - filtertype: age
      direction: older
      unit: days
      unit_count: 30

# 範例 2：安全的 space filter
- action: delete_indices
  description: "Keep only 100GB of application logs"
  filters:
    - filtertype: pattern
      kind: prefix
      value: app-logs-       # ✅ 先用 pattern 限制範圍
    - filtertype: space
      disk_space: 100         # ✅ 再使用 space filter
```

### ❌ 危險的配置範例

```yaml
# ❌ 危險 1：沒有 pattern filter
- action: delete_indices
  filters:
    - filtertype: age
      direction: older
      unit: days
      unit_count: 30
  # ⚠️ 會選中所有超過 30 天的 index，包括系統 index！

# ❌ 危險 2：過於寬鬆的 regex
- action: delete_indices
  filters:
    - filtertype: pattern
      kind: regex
      value: ".*"            # 匹配所有 index
  # ⚠️ 雖然系統 index 會被排除（修復後），但仍然太危險

# ❌ 危險 3：單獨使用 space filter
- action: delete_indices
  filters:
    - filtertype: space
      disk_space: 100
  # ⚠️ 會選中所有 index！（修復後會被阻止）
```

### 🔒 最佳實踐

1. **Always use pattern filter first**
   - 先用 pattern 限制範圍
   - 再用 age/space 進一步篩選

2. **Always test in test_mode first**
   ```yaml
   # setting.yml
   test_mode: true
   ```
   - 檢查日誌中選中的 index 列表
   - 確認沒有意外選中的 index

3. **Use restrictive patterns**
   - ✅ `logs-myapp-*` (specific prefix)
   - ✅ `myapp-*-2024*` (specific pattern)
   - ❌ `.*` (too broad)
   - ❌ `.` (would match system indices)

4. **Double-check delete_indices actions**
   - 這是唯一不可逆的操作
   - 建議使用 close 代替 delete 作為第一步
